    function createStars() {
        const container = document.getElementById('stars');
        container.innerHTML = '';
        const count = currentTheme === 'space' ? 140 : 80;
        for (let i = 0; i < count; i++) {
            const star = document.createElement('div');
            star.className = 'star';
            star.style.left = Math.random() * 100 + '%';
            star.style.top = Math.random() * 100 + '%';
            star.style.width = (Math.random() * 3 + 1) + 'px';
            star.style.height = star.style.width;
            star.style.animationDelay = Math.random() * 3 + 's';
            star.style.animationDuration = (Math.random() * 3 + 2) + 's';
            container.appendChild(star);
        }
    }

    function createShootingStar() {
        if (currentTheme !== 'space') return;
        const star = document.createElement('div');
        star.className = 'shooting-star';
        star.style.position = 'fixed';
        star.style.width = '1.5px';
        star.style.height = '70px';
        star.style.background = 'linear-gradient(to bottom, white, transparent)';
        star.style.pointerEvents = 'none';
        star.style.zIndex = '0';
        star.style.top = Math.random() * 25 + '%';
        star.style.left = Math.random() * 70 + '%';
        star.style.opacity = '0';
        star.style.transform = 'rotate(-45deg)';
        document.body.appendChild(star);
        
        let start = performance.now();
        function anim(time) {
            let p = (time - start) / 1500;
            if (p < 0.1) star.style.opacity = p * 10;
            else if (p < 0.5) { star.style.opacity = 1; star.style.transform = `translateY(${p*400}px) rotate(-45deg)`; }
            else { star.style.opacity = 1 - (p-0.5)*2; star.style.transform = `translateY(400px) rotate(-45deg)`; }
            if (p < 1) requestAnimationFrame(anim);
            else star.remove();
        }
        requestAnimationFrame(anim);
    }

        playCorrect() {
            this.init(); const ctx = this.audioContext; const now = ctx.currentTime;
            const osc = ctx.createOscillator(), gain = ctx.createGain();
            osc.connect(gain); gain.connect(ctx.destination);
            osc.type = 'sine';
            osc.frequency.setValueAtTime(523, now);
            osc.frequency.linearRampToValueAtTime(784, now + 0.15);
            gain.gain.setValueAtTime(0.3, now);

            const cq = state.editingChallenge.questions[state.editingQuestionIndex];
            let nav = '';
            if (qc > 0) { nav = `<div class="question-nav">`; for (let i=0;i<qc;i++) nav += `<button class="${i===state.editingQuestionIndex?'':'secondary'}" onclick="selectQuestion(${i})">${i+1}</button>`; nav += `</div>`; }
            let form = '';
            if (cq) form = `<div class="form-group"><label>السؤال ${state.editingQuestionIndex+1}</label><textarea id="qText" placeholder="اكتب سؤالك هنا...">${escapeHtml(cq.text)}</textarea></div><div class="form-group"><label>الخيارات - اضغط على الدائرة لتحديد الإجابة الصحيحة</label><div class="options-grid" id="optionsContainer"></div></div>`;
            app.innerHTML = `
                <div class="card">
                    <h1>✏️ إنشاء تحدي</h1>
                    <div class="form-group"><label>عنوان التحدي (اختياري)</label><input type="text" id="challengeTitle" placeholder="مثلاً: تحدي محمد" value="${escapeHtml(state.editingChallenge.title)}"></div>
                    <div class="form-group"><label>عدد الأسئلة</label><select id="questionCount" onchange="changeQuestionCount(this.value)">${[3,4,5,6,7,8,9,10].map(n=>`<option value="${n}" ${state.questionCount===n?'selected':''}>${n} أسئلة</option>`).join('')}</select></div>
                    <p style="text-align:center;margin:16px 0;font-weight:600;font-size:18px;color:#FFD700;">الأسئلة: ${qc} / ${state.questionCount}</p>
                    ${nav}${form}
                    ${qc<state.questionCount?`<button onclick="addQuestion()" class="secondary">➕ إضافة سؤال</button>`:''}
                    ${qc===state.questionCount&&qc>=3?`<button onclick="saveChallenge()">🚀 إنشاء التحدي</button>`:''}
                    <button onclick="goHome()" class="secondary" style="margin-top:8px;">رجوع</button>
                </div>`;
            if (cq) {
                const cont = document.getElementById('optionsContainer');
                if (cont) for (let i=0;i<4;i++) {
                    const div = document.createElement('div');
                    div.className = `option-item ${cq.correct===i?'selected':''}`;
                    div.onclick = function(){selectCorrect(i);};
                    div.innerHTML = `<div class="option-radio"></div><input type="text" id="opt${i}" value="${escapeHtml(cq.options[i])}" placeholder="الخيار ${i+1}" onclick="event.stopPropagation()" oninput="updateOption(${i},this.value)" style="flex:1;border:none;background:transparent;padding:0;font-size:16px;color:white;">`;
                    cont.appendChild(div);
                }
                const qt = document.getElementById('qText'); if(qt) qt.oninput = function(){state.editingChallenge.questions[state.editingQuestionIndex].text=this.value;};
                const ti = document.getElementById('challengeTitle'); if(ti) ti.oninput = function(){state.editingChallenge.title=this.value;};
            }
            return;
        }

        if (state.page === 'share') {
            app.innerHTML = `
                <div class="card">
                    <h1>🎉 تم إنشاء التحدي!</h1>
                    <p style="text-align:center;margin:16px 0;color:rgba(255,255,255,0.7);">انسخ الرابط وشاركه مع أصدقائك:</p>
                    <div class="link-box">${state.shareLink}</div>
                    <button onclick="copyLink()">📋 نسخ الرابط</button>
                    <div class="share-buttons">
                        <button class="share-btn share-whatsapp" onclick="shareWhatsApp('${state.shareLink.replace(/'/g,"\\'")}')">💬 واتساب</button>
                        <button class="share-btn share-telegram" onclick="shareTelegram('${state.shareLink.replace(/'/g,"\\'")}')">✈️ تليغرام</button>
                    </div>
                    <button onclick="goHome()" class="secondary" style="margin-top:8px;">العودة للرئيسية</button>
                </div>`; return;
        }

        if (state.page === 'name_entry') {
            const t = state.challenge.questions.length;
            app.innerHTML = `
                <div class="card">
                    <h1>👋 مرحباً بك!</h1>
                    <p style="text-align:center;margin:16px 0;font-size:18px;">${escapeHtml(state.challenge.title||'تحدي جديد')}</p>
                    <p style="text-align:center;color:rgba(255,255,255,0.5);">${t} أسئلة</p>
                    <div class="form-group"><label>ما اسمك؟</label><input type="text" id="playerNameInput" placeholder="اكتب اسمك هنا..."></div>
                    <button onclick="startQuiz()">🚀 ابدأ التحدي</button>
                </div>`; return;
        }

        if (state.page === 'quiz') {
            const q = state.challenge.questions[state.questionIndex];
            const t = state.challenge.questions.length;
            const prog = (state.questionIndex/t)*100;
            app.innerHTML = `
                <div class="card">
                    <div class="progress-bar"><div class="progress-fill" style="width:${prog}%"></div></div>
                    <p class="progress-text">السؤال ${state.questionIndex+1} من ${t}</p>
                    <p class="question-text">${escapeHtml(q.text)}</p>
                    <div class="options-grid" id="quizOptions"></div>
                </div>`;
            const cont = document.getElementById('quizOptions');
            if (cont) for (let i=0;i<q.options.length;i++) {
                const div = document.createElement('div');
                div.className = 'option-item';
                div.onclick = function(){answerQuestion(i);};
                div.innerHTML = `<div class="option-radio"></div><span>${escapeHtml(q.options[i])}</span>`;
                cont.appendChild(div);
            }
            return;
        }

        if (state.page === 'feedback') {
            const q = state.challenge.questions[state.questionIndex];
            const la = state.answers[state.answers.length-1];
            const t = state.challenge.questions.length;
            const prog = ((state.questionIndex+1)/t)*100;
            let opts = '';
            for (let i=0;i<q.options.length;i++) {
                let cls='',mark='';
                if (i===q.correct){cls='correct';mark='<span class="correct-mark">✓</span>';}
                else if(i===la.selected&&!la.isCorrect){cls='wrong';mark='<span class="wrong-mark">✗</span>';}
                opts += `<div class="option-item ${cls}">${mark}<span>${escapeHtml(q.options[i])}</span></div>`;
            }
            app.innerHTML = `
                <div class="card">
                    <div class="progress-bar"><div class="progress-fill" style="width:${prog}%"></div></div>
                    <p class="progress-text">السؤال ${state.questionIndex+1} من ${t}</p>
                    <p class="question-text">${escapeHtml(q.text)}</p>
                    <div class="options-grid">${opts}</div>
                    <div class="feedback" style="background:${la.isCorrect?'rgba(16,185,129,0.2)':'rgba(239,68,68,0.2)'};color:${la.isCorrect?'#10B981':'#EF4444'}">${la.isCorrect?'✅ إجابة صحيحة!':'❌ إجابة خاطئة'}</div>
                </div>`; return;
        }

        if (state.page === 'result') {
            const t = state.challenge.questions.length;
            const ts = state.results[0]?.score||0;
            const title = getTitle(state.score, t);
            let lb = '';
            for (const r of state.results) lb += `<div class="leaderboard-item ${r.score===ts?'top':''}"><span class="player-name">${r.score===ts?'<span class="crown">👑</span>':''}${escapeHtml(r.playerName)}</span><span class="player-score">${r.score} / ${t}</span></div>`;
            app.innerHTML = `
                <div class="card">
                    <h1>🎉 أحسنت يا ${escapeHtml(state.playerName)}!</h1>
                    <div class="result-score">${state.score} / ${t}</div>
                    <p class="result-message">${title}</p>
                    <div class="leaderboard"><h3>🏆 لوحة المتصدرين</h3>${lb||'<p style="text-align:center;color:rgba(255,255,255,0.5);">كن أول من يلعب!</p>'}</div>
                    <button onclick="goHome()" style="margin-top:20px;">العودة للرئيسية</button>
                </div>`;
            if ((state.score/t)>=0.8) setTimeout(()=>launchConfetti(),300);
            return;
        }

        if (state.page === 'notfound') { app.innerHTML = `<div class="card"><h1>😕</h1><p style="text-align:center;margin:16px 0;">التحدي غير موجود</p><button onclick="goHome()">العودة للرئيسية</button></div>`; return; }
        if (state.page === 'already_completed') { app.innerHTML = `<div class="card"><h1>😊</h1><p style="text-align:center;margin:16px 0;">لقد أكملت هذا التحدي مسبقاً</p><button onclick="goHome()">العودة للرئيسية</button></div>`; return; }
    }

    function escapeHtml(t) { const d = document.createElement('div'); d.textContent = t; return d.innerHTML; }
    function goHome() { state.page = 'home'; window.location.hash = ''; render(); }
    function goToCreate() { state.page = 'create'; state.questionCount = 10; state.editingChallenge = { title: '', questions: [] }; state.editingQuestionIndex = 0; window.location.hash = ''; render(); }
    function changeQuestionCount(c) { c=parseInt(c); state.questionCount=c; while(state.editingChallenge.questions.length>c) state.editingChallenge.questions.pop(); if(state.editingQuestionIndex>=c) state.editingQuestionIndex=Math.max(0,c-1); render(); }
    function addQuestion() { if(state.editingChallenge.questions.length>=state.questionCount) return; state.editingChallenge.questions.push({text:'',options:['','','',''],correct:0}); state.editingQuestionIndex=state.editingChallenge.questions.length-1; render(); }
    function selectQuestion(i) { state.editingQuestionIndex=i; render(); }
    function selectCorrect(i) { state.editingChallenge.questions[state.editingQuestionIndex].correct=i; render(); }
    function updateOption(i,v) { state.editingChallenge.questions[state.editingQuestionIndex].options[i]=v; }

    async function saveChallenge() {
        const ch = state.editingChallenge;
        for (let i=0;i<ch.questions.length;i++) { if(!ch.questions[i].text.trim()){alert('الرجاء كتابة نص السؤال '+(i+1));return;} for(let j=0;j<ch.questions[i].options.length;j++) if(!ch.questions[i].options[j].trim()){alert('الرجاء كتابة جميع الخيارات للسؤال '+(i+1));return;} }
        try {
            const dr = await db.collection('challenges').add({title:ch.title,questions:ch.questions,questionCount:ch.questions.length,createdAt:new Date().toISOString()});
            state.shareLink = window.location.href.split('#')[0] + '#/c/' + dr.id;
            state.page = 'share'; render();
        } catch(e) { alert('حدث خطأ: '+e.message); }
    }

    function copyLink() { navigator.clipboard.writeText(state.shareLink).then(()=>alert('تم النسخ! ✅')).catch(()=>prompt('انسخ الرابط:',state.shareLink)); }
    function startQuiz() { state.playerName = document.getElementById('playerNameInput')?.value?.trim()||'مجهول'; state.questionIndex=0; state.answers=[]; state.page='quiz'; render(); }

    function answerQuestion(s) {
        const q = state.challenge.questions[state.questionIndex];
        const ic = s === q.correct;
        ic ? AudioSystem.playCorrect() : AudioSystem.playWrong();
        state.answers.push({selected:s,isCorrect:ic});
        state.page='feedback'; render();
        setTimeout(()=>{
            if(state.questionIndex<state.challenge.questions.length-1){state.questionIndex++;state.page='quiz';}
            else finishQuiz();
            render();
        },1500);
    }

    function finishQuiz() {
        state.score = state.answers.filter(a=>a.isCorrect).length;
        const res = getResults(state.challengeId);
        res.push({playerName:state.playerName,score:state.score,date:new Date().toISOString()});
        res.sort((a,b)=>b.score-a.score);
        saveResults(state.challengeId,res); state.results=res;
        markCompleted(state.challengeId);
        state.page='result'; render();
    }

    async function loadFromHash() {
        const hash = window.location.hash;
        if (hash.startsWith('#/c/')) {
            const cid = hash.replace('#/c/','');
            if (isCompleted(cid)) { state.page='already_completed'; render(); return; }
            state.page='loading'; render();
            try {
                const doc = await db.collection('challenges').doc(cid).get();
                if (!doc.exists) { state.page='notfound'; render(); return; }
                state.challenge = doc.data(); state.challengeId = cid;
                state.page='name_entry'; render();
            } catch(e) { state.page='notfound'; render(); }
            return;
        }
        goHome();
    }

    // شهب كل 15 ثانية في الفضاء
    setInterval(() => { if (currentTheme==='space' && Math.random()>0.5) createShootingStar(); }, 15000);

    window.addEventListener('hashchange', loadFromHash);
    loadFromHash();
</script>
